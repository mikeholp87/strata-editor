cmake_minimum_required(VERSION 3.22)
project(StrataEditor VERSION 0.1.2 LANGUAGES CXX)
set(STRATA_PRODUCT_NAME "Strata Editor" CACHE STRING "Application display name")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent PrintSupport)
add_library(strata STATIC
 src/core/document.cpp src/core/recovery.cpp src/core/persistence.cpp src/core/export.cpp src/core/shortcuts.cpp
 src/editor/codeeditor.cpp src/editor/codeeditor_markdown.cpp src/editor/completion.cpp src/editor/highlighter.cpp
 src/ui/documentview.cpp src/ui/mainwindow.cpp src/ui/projectsearch.cpp src/ui/gitpanel.cpp
 src/ui/mainwindow_commands.cpp src/ui/mainwindow_writing.cpp
 src/ui/mainwindow_session.cpp src/ui/mainwindow_navigation.cpp src/ui/shortcutdialog.cpp src/ui/windowstate.cpp)
target_include_directories(strata PUBLIC src)
target_compile_definitions(strata PUBLIC STRATA_PRODUCT_NAME="${STRATA_PRODUCT_NAME}")
target_link_libraries(strata PUBLIC Qt6::Widgets Qt6::Concurrent Qt6::PrintSupport)
if(MSVC)
 target_compile_options(strata PRIVATE /W4)
else()
 target_compile_options(strata PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(strata-editor src/main.cpp resources/resources.qrc)
target_link_libraries(strata-editor PRIVATE strata)
option(STRATA_BUILD_BENCHMARKS "Build local measurement tool" OFF)
if(STRATA_BUILD_BENCHMARKS)
 add_executable(strata-measure benchmarks/measure.cpp)
 target_link_libraries(strata-measure PRIVATE strata)
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
 add_executable(strata-tests tests/test_strata.cpp resources/resources.qrc)
 target_link_libraries(strata-tests PRIVATE strata Qt6::Test)
 add_executable(strata-shortcut-tests tests/test_shortcuts.cpp resources/resources.qrc)
 target_link_libraries(strata-shortcut-tests PRIVATE strata Qt6::Test)
 add_executable(strata-audit-tests tests/test_audit.cpp resources/resources.qrc)
 target_link_libraries(strata-audit-tests PRIVATE strata Qt6::Test)
 add_executable(strata-fake-git tests/fake_git.cpp)
 target_link_libraries(strata-fake-git PRIVATE Qt6::Core)
 add_executable(strata-hardening-tests tests/test_hardening.cpp resources/resources.qrc)
 target_link_libraries(strata-hardening-tests PRIVATE strata Qt6::Test)
 target_compile_definitions(strata-hardening-tests PRIVATE STRATA_FAKE_GIT_PATH="$<TARGET_FILE:strata-fake-git>")
 add_dependencies(strata-hardening-tests strata-fake-git)
 add_executable(strata-editing-tests tests/test_editing.cpp)
 target_link_libraries(strata-editing-tests PRIVATE strata Qt6::Test)
 add_test(NAME strata-editing COMMAND strata-editing-tests)
 set_tests_properties(strata-editing PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_executable(strata-usability-tests tests/test_usability.cpp resources/resources.qrc)
 target_link_libraries(strata-usability-tests PRIVATE strata Qt6::Test)
 add_test(NAME strata-usability COMMAND strata-usability-tests)
 set_tests_properties(strata-usability PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME strata-hardening COMMAND strata-hardening-tests)
 set_tests_properties(strata-hardening PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME strata-audit COMMAND strata-audit-tests)
 set_tests_properties(strata-audit PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME strata-shortcuts COMMAND strata-shortcut-tests)
 set_tests_properties(strata-shortcuts PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME strata-behavior COMMAND strata-tests)
 set_tests_properties(strata-behavior PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS strata-editor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/linux/strata-editor.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/strata-editor.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/strata-editor)
set(CPACK_PACKAGE_NAME strata-editor)
set(CPACK_PACKAGE_CONTACT "Mike Holp <4627227+mikeholp87@users.noreply.github.com>")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native text, code, Markdown and HTML editor")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/mikeholp87/strata-editor")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
 OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION editors)
# Hand-written Depends: shlibdeps on Ubuntu 24.04 names the *t64 Qt packages, which
# Debian does not have. Accept either spelling so one .deb installs on both.
set(CPACK_DEBIAN_PACKAGE_DEPENDS
    "libc6 (>= 2.34), libgcc-s1 (>= 3.0), libstdc++6 (>= 11), qt6-qpa-plugins, libqt6core6t64 (>= 6.4.0) | libqt6core6 (>= 6.4.0), libqt6gui6t64 (>= 6.4.0) | libqt6gui6 (>= 6.4.0), libqt6widgets6t64 (>= 6.4.0) | libqt6widgets6 (>= 6.4.0), libqt6printsupport6t64 (>= 6.1.2) | libqt6printsupport6 (>= 6.1.2)")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS OFF)
set(CPACK_GENERATOR "TGZ;DEB")
include(CPack)
